1use super::conv::*;

3#[repr(u8)]
4#[derive(Clone, Copy)]
5#[derive(PartialEq)]
6pub enum BinaryOperator {
7    // Signed integer arithmetic
8    Add = 0, Sub = 1, SubR = 2,
9    Mul = 3, Div = 4, Mod = 5, 
10    Min = 6, Max = 7,
11    // Floating point arithmetic
12    AddF = 8, SubF = 9, SubFR = 10,
13    MulF = 11, DivF = 12, ModF = 13,
14    MinF = 14, MaxF = 15,
15    // Bitwise arithmetic
16    And = 16, Or = 17, Xor = 18, Xnor = 19,
17    // Unsigned integers
18    MinU = 20, MaxU = 21, DivU = 22, ModU = 23,
19    // Rots/Shifts
20    Rol = 24, Ror = 25,
21    Lsl = 26, Lsr = 27, Asr = 28,
22    // Special
23    Fpintexp = 29,
24    Nop = 31,
25}
26
27impl BinaryOperator {
28    pub fn compute(&self, lhs: u32, rhs: u32) -> u32 {
29        match self {
30            // Signed int arithmetic
31            BinaryOperator::Add => f_i32(i_i32(lhs) + i_i32(rhs)),
32            BinaryOperator::Sub => f_i32(i_i32(lhs) - i_i32(rhs)),
33            BinaryOperator::SubR => f_i32(i_i32(rhs) - i_i32(lhs)),
34            BinaryOperator::Mul => f_i32(i_i32(lhs) * i_i32(rhs)),
35            BinaryOperator::Div => f_i32(i_i32(lhs) / i_i32(rhs)),
36            BinaryOperator::Mod => f_i32(i_i32(lhs) % i_i32(rhs)),
37            BinaryOperator::Min => f_i32(i_i32(lhs).min(i_i32(rhs))),
38            BinaryOperator::Max => f_i32(i_i32(lhs).max(i_i32(rhs))),
39
40            // Floating point math
41            BinaryOperator::AddF => f_f32(i_f32(lhs) + i_f32(rhs)),
42            BinaryOperator::SubF => f_f32(i_f32(lhs) - i_f32(rhs)),
43            BinaryOperator::SubFR => f_f32(i_f32(rhs) - i_f32(lhs)),
44            BinaryOperator::MulF => f_f32(i_f32(lhs) * i_f32(rhs)),
45            BinaryOperator::DivF => f_f32(i_f32(lhs) / i_f32(rhs)),
46            BinaryOperator::ModF => f_f32(i_f32(lhs) % i_f32(rhs)),
47            BinaryOperator::MinF => f_f32(i_f32(lhs).min(i_f32(rhs))),
48            BinaryOperator::MaxF => f_f32(i_f32(lhs).max(i_f32(rhs))),
49
50            // Bitwise math
51            BinaryOperator::And => lhs & rhs,
52            BinaryOperator::Or => lhs | rhs,
53            BinaryOperator::Xor => lhs ^ rhs,
54            BinaryOperator::Xnor => !(lhs ^ rhs),
55            BinaryOperator::MinU => lhs.min(rhs),
56            BinaryOperator::MaxU => lhs.max(rhs),
57            BinaryOperator::DivU => lhs / rhs,
58            BinaryOperator::ModU => lhs % rhs,
59
60            // Shifts/other
61            BinaryOperator::Rol => lhs.rotate_left(rhs),
62            BinaryOperator::Ror => lhs.rotate_right(rhs),
63            BinaryOperator::Lsl => lhs >> rhs,
64            BinaryOperator::Lsr => lhs << rhs,
65            BinaryOperator::Asr => f_i32(i_i32(lhs) >> i_i32(rhs)),
66            BinaryOperator::Fpintexp => f_f32(i_f32(lhs) * (2f32).powi(i_i32(rhs))),
67
68            BinaryOperator::Nop => 0,
69        }
70    }
71
72    pub fn decode(op: u8) -> BinaryOperator {
73        if op <= 29 {
74            unsafe { std::mem::transmute(op) }
75        } else if op <= 31 {
76            BinaryOperator::Nop
77        } else {
78            panic!("Invalid operator!");
79        }
80    }
81}
