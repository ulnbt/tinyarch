7use std::fmt;
8use super::{BinaryOperator, Condition, Instruction};
9use fmt_reg::{reg_name, r_rix_name};

12impl fmt::Display for Instruction {
13    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
14        match self {
15            Instruction::RawData(val) => write!(f, "DAT {val:#x}"),
16
17            Instruction::MvRg { src, dst, src_m, dst_m } =>
18                write!(f, "{} -> {}", r_rix_name(*src, *src_m), r_rix_name(*dst, *dst_m)),
19
20            Instruction::MvRd { dst, val } =>
21                write!(f, "@{:#x} -> {}", val, r_rix_name(*dst, 1)),
22
23            Instruction::MvLd { dst, val } =>
24                write!(f, "{:#x} -> {}", val, r_rix_name(*dst, 1)),
25
26            // (behaviour might change!)
27            Instruction::MvLf { dst, val } =>
28                write!(f, "{} -> r{:X}", f32::from_bits(*val), dst),
29
30            Instruction::OpIm { op, lhs, dst, rhs } =>
31                write!(f, "{} {} {:#x} -> {}", reg_name(*lhs), op, rhs, reg_name(*dst)),
32
33            Instruction::OpRg { op, lhs, rhs, dst } =>
34                write!(f, "{} {} {} -> {}",
35                    r_rix_name(*lhs, 1), op, r_rix_name(*rhs, 1), r_rix_name(*dst, 1)),
36
37            Instruction::BrCoIm { co, lhs, rhs, val } =>
38                write!(f, "if {} {} {} jmp {:#x}", reg_name(*lhs), co, reg_name(*rhs), val),
39
40            // Register: all three are r_rix bytes.
41            Instruction::BrCoRg { co, lhs, rhs, dst } =>
42                write!(f, "if {} {} {} jmp {}",
43                    r_rix_name(*lhs, 1), co, r_rix_name(*rhs, 1), r_rix_name(*dst, 1)),
44
45            // Flag-based branches (NOT finalized)
46            Instruction::BrFlagIm { flag, val } =>
47                write!(f, "if flag({}) jmp {:#x}", flag, val),
48            Instruction::BrFlagRg { flat, dst } =>
49                write!(f, "if flag({}) jmp {}", flat, reg_name(*dst)),
50            
51            Instruction::BrUnconditional { dst } =>
52                write!(f, "jmp {:#x}", *dst),
53
54            // Stack
55            Instruction::SpStackPos { val } => write!(f, "stack {val}"),
56
57            Instruction::StPushRg { esc, src, src_val } => 
58                if *src_val == 0 { write!(f, "push {}", r_rix_name(*src, 1)) }
59                else { write!(f, "push {} <- {}", r_rix_name(*src, 1), r_rix_name(*src_val, 1)) },
60            Instruction::StPushRgImm { esc, src, val } =>
61                write!(f, "push {} <- {val:#x}", r_rix_name(*src, 1)),
62            Instruction::StPushPcI { esc, sparam, val } => 
63                if *sparam == 0 { write!(f, "sjmp {val:#x}") } else { write!(f, "sjmp {val:#x} ({sparam})") },
64            Instruction::StPushPcR { esc, sparam, dst } =>
65                if *sparam == 0 { write!(f, "sjmp {}", r_rix_name(*dst, 1)) } else { write!(f, "sjmp {} ({sparam})", r_rix_name(*dst, 1)) },
66            Instruction::StPushStI { esc, sparam, val } => 
67                if *sparam == 0 { write!(f, "call {val:#x}") } else { write!(f, "call {val:#x} ({sparam})") },
68            Instruction::StPushStR { esc, sparam, dst } =>
69                if *sparam == 0 { write!(f, "call {}", r_rix_name(*dst, 1)) } else { write!(f, "call {} ({sparam})", r_rix_name(*dst, 1)) },
70
71            Instruction::StPeek { esc, dst, offset_rg, offset } => {
72                if *offset_rg == 0 && *offset == 0 { 
73                    write!(f, "peek {} + {} -> {}", r_rix_name(*offset_rg, 1), *offset, r_rix_name(*dst, 1))  
74                } else if *offset == 0 { 
75                    write!(f, "peek {} -> {}", r_rix_name(*offset_rg, 1), r_rix_name(*dst, 1))
76                } else if *offset_rg == 0 {
77                    write!(f, "peek {} -> {}", *offset, r_rix_name(*dst, 1))
78                } else {
79                    write!(f, "peek {}", r_rix_name(*dst, 1))
80                }
81            }
82            Instruction::StPopRg { esc, dst } => write!(f, "pop {}", r_rix_name(*dst, 1)),
83            Instruction::StPopPc { esc, sparam } => if *sparam == 0 {write!(f, "sret")} else {write!(f, "sret ({sparam})")},
84            Instruction::StPopSt { esc, sparam } => if *sparam == 0 {write!(f, "ret")} else {write!(f, "ret ({sparam})")},
85
86            // other special
87            Instruction::SpHalt          => write!(f, "halt"),
88            Instruction::SpNop           => write!(f, "nop"),
89
90            Instruction::SpIntRet        => write!(f, "intret"),
91            Instruction::SpInt           => write!(f, "int"),
92            Instruction::SpVec           => write!(f, "vec"),
93            Instruction::SpWrith         => write!(f, "writh"),
94            Instruction::SpReath         => write!(f, "reath"),
95            Instruction::SpInith         => write!(f, "inith"),
96            Instruction::SpUmem          => write!(f, "umem"),
97            Instruction::SpUser          => write!(f, "user"),
98            Instruction::SpDebug         => write!(f, "debug"),
99        }
100    }
101}
