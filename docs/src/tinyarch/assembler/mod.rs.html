27/// Parses a multiline program of assembly into an `ExecutableWriter`
28pub fn assemble<T: ExecutableWriter>(src: &str, mut ew: &mut T) -> Result<(), String> {
29    let mut instructions: Vec<String> = Vec::new();
30    let ew_macros = T::macros();
31
32    // pass 1: strip comments, find labels, trim whitespace, apply macros
33    for line in src.lines() {
34        // Strip comments
35        let line = line.split(';').next().unwrap().trim();
36        if line.is_empty() { continue; }
37
38        // Find labels
39        let instr: &str = if let Some(pos) = line.find(':') {
40            let label = line[..pos].trim();
41            if !label.is_empty() && label.chars().all(|c| c.is_alphanumeric() || c == '_') {
42                ew.add_label(label, instructions.len() as u32)?;
43                line[pos + 1..].trim()
44            } else { line }
45        } else { line };
46
47        if instr.is_empty() { continue; }
48
49        // Apply default macros
50        for (n, f) in DEFAULT_ASSEMBLER_MACROS {
51            if instr.starts_with(n) { 
52                f(n, &mut instructions)?;
53                continue;
54            }
55        }
56
57        // Apply executable writer's macros
58        for (n, f) in ew_macros {
59            if instr.starts_with(n) { 
60                f(&mut ew, n, &mut instructions)?;
61                continue;
62            }
63        }
64
65        // Otherwise push instruction as is
66        instructions.push(instr.to_string());
67    }
68
69    // pass 2: replace labels with numerical addresses in instructions
70
71    // get keys and sort by length (as to not lazily match label substrings)
72    let mut labels = ew.get_labels()?;    
73    labels.sort_by_key(|a| std::cmp::Reverse(a.len()));
74
75    for label in labels {
76        let value = format!("{}", ew.get_label_pos(label)?);
77        let at_label = format!("@{}", label);
78        let at_value = format!("@{}", value);
79
80        for line in instructions.iter_mut() {
81            // The find/replace needs to be much more robust I feel.
82            // Am I sure that the word boundary is not needed for at_label's?
83            *line = line.replace(&at_label, &at_value);
84            if line.contains(label) {
85                *line = replace_whole_word(line, label, &value); // attempts to replace it
86            }
87        }
88    }
89
90    // pass 3: encode into writter
91    for (pos, instr) in instructions.iter().enumerate() {
92        println!("{instr}");
93        match parse::parse_instruction(instr) {
94            Ok(n) => { 
95                println!("==> {}", n);
96                ew.write_instruction(n.encode(), pos)?; 
97            }
98            Err(e) => { return Err(format!( "Assembly error at instruction: {e}" )) }
99        }
100    }
101
102    Ok(())
103}

105fn replace_whole_word(s: &str, from: &str, to: &str) -> String {
106    if from.is_empty() {
107        return s.to_string();
108    }
109
110    let mut result = String::with_capacity(s.len());
111    let mut last_end = 0;
112    for (start, matched) in s.match_indices(from) {
113        let end = start + matched.len();
114
115        let left_ok = s[..start].chars().next_back()
116            .map_or(true, |c| !c.is_alphanumeric() && c != '_');
117        let right_ok = s[end..].chars().next()
118            .map_or(true, |c| !c.is_alphanumeric() && c != '_');
119
120        result.push_str(&s[last_end..start]);
121
122        if left_ok && right_ok { result.push_str(to); } 
123        else { result.push_str(matched); }
124
125        last_end = end;
126    }
127
128    result.push_str(&s[last_end..]);
129    result
130}
