11impl<'a> SimpleExecutableWriter<'a> {

17impl<'a> ExecutableWriter for SimpleExecutableWriter<'a> {
18    fn write_instruction(&mut self, instr: u32, pos: usize) -> Result<(), String> {
19        self.machine.memory[pos + self.offset] = instr;
20        Ok(())
21    }
22    
23    fn add_label(&mut self, label: &str, pos: u32) -> Result<(), String> {
24        if self.labels.insert(label.to_string(), pos + self.offset as u32).is_some() {
25            Err(format!("Label {} defined twice?", label))
26        } else { Ok(()) }
27    }
28    
29    fn get_labels(&self) -> Result<Vec<&str>, String> {
30        Ok(self.labels.keys().map(|s| &s[..]).collect())
31    }
32    
33    fn get_label_pos(&self, label: &str) -> Result<u32, String> {
34        // the labels are taken directly from get_labels so it should be fine.
35        Ok(*self.labels.get(label).expect("Executable writer: label not found!")) 
36    }
